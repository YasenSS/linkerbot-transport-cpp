cmake_minimum_required(VERSION 3.16)
project(comm-shared-consumer LANGUAGES CXX)
find_package(linkerbot-comm 0.3 CONFIG REQUIRED)
add_library(shared-consumer SHARED shared.cpp)
target_link_libraries(shared-consumer PRIVATE linkerbot::comm)
add_executable(shared-consumer-smoke main.cpp)
target_link_libraries(shared-consumer-smoke PRIVATE shared-consumer)
enable_testing()
add_test(NAME shared-consumer-smoke COMMAND shared-consumer-smoke)
set_tests_properties(shared-consumer-smoke PROPERTIES TIMEOUT 10)
