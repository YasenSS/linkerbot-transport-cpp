cmake_minimum_required(VERSION 3.16)
project(linkerbot_transport_ros1_smoke LANGUAGES CXX)
find_package(catkin REQUIRED COMPONENTS roscpp nodelet pluginlib)
find_package(linkerbot-transport 0.3 CONFIG REQUIRED)
catkin_package(LIBRARIES transport_smoke_nodelet CATKIN_DEPENDS roscpp nodelet pluginlib)
add_library(transport_smoke_nodelet SHARED src/nodelet.cpp)
target_include_directories(transport_smoke_nodelet SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
target_compile_features(transport_smoke_nodelet PRIVATE cxx_std_17)
target_link_libraries(transport_smoke_nodelet PRIVATE linkerbot::can ${catkin_LIBRARIES})
add_executable(transport_smoke_node src/node.cpp)
target_include_directories(transport_smoke_node SYSTEM PRIVATE ${catkin_INCLUDE_DIRS})
target_compile_features(transport_smoke_node PRIVATE cxx_std_17)
target_link_libraries(transport_smoke_node PRIVATE linkerbot::can ${catkin_LIBRARIES})
# External ROS headers are system includes; authored node/nodelet code is strict.
foreach(smoke_target transport_smoke_node transport_smoke_nodelet)
  target_compile_options(${smoke_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
endforeach()
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(launch/smoke.test)
endif()
