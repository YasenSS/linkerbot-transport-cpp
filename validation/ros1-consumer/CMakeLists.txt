cmake_minimum_required(VERSION 3.16)
project(linkerbot_comm_ros1_smoke LANGUAGES CXX)
find_package(catkin REQUIRED COMPONENTS roscpp nodelet pluginlib)
find_package(linkerbot-comm 0.3 CONFIG REQUIRED)
catkin_package(LIBRARIES comm_smoke_nodelet CATKIN_DEPENDS roscpp nodelet pluginlib)
add_library(comm_smoke_nodelet SHARED src/nodelet.cpp)
target_include_directories(comm_smoke_nodelet PRIVATE ${catkin_INCLUDE_DIRS})
target_compile_features(comm_smoke_nodelet PRIVATE cxx_std_17)
target_link_libraries(comm_smoke_nodelet PRIVATE linkerbot::comm ${catkin_LIBRARIES})
add_executable(comm_smoke_node src/node.cpp)
target_include_directories(comm_smoke_node PRIVATE ${catkin_INCLUDE_DIRS})
target_compile_features(comm_smoke_node PRIVATE cxx_std_17)
target_link_libraries(comm_smoke_node PRIVATE linkerbot::comm ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  add_rostest(launch/smoke.test)
endif()
